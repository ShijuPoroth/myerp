# Database Schema for MWH Management ERP

## Locations
- id (INTEGER PRIMARY KEY)
- name (TEXT)
- address (TEXT)
- type (TEXT: 'equipment', 'kitchen', 'uniform')
- created_at (DATETIME)

## Equipment
- id (INTEGER PRIMARY KEY)
- name (TEXT)
- serial_number (TEXT)
- category (TEXT)
- model (TEXT)
- purchase_date (DATE)
- purchase_cost (DECIMAL)
- location_id (INTEGER FOREIGN KEY)
- status (TEXT: 'active', 'maintenance', 'retired')
- photo_path (TEXT)
- description (TEXT)
- created_at (DATETIME)
- updated_at (DATETIME)

## Equipment_Transfers
- id (INTEGER PRIMARY KEY)
- equipment_id (INTEGER FOREIGN KEY)
- from_location_id (INTEGER FOREIGN KEY)
- to_location_id (INTEGER FOREIGN KEY)
- transfer_date (DATETIME)
- notes (TEXT)
- created_by (TEXT)
- created_at (DATETIME)

## Maintenance_Logs
- id (INTEGER PRIMARY KEY)
- equipment_id (INTEGER FOREIGN KEY)
- maintenance_type (TEXT: 'preventive', 'corrective', 'emergency')
- description (TEXT)
- cost (DECIMAL)
- performed_by (TEXT)
- performed_date (DATE)
- next_maintenance_date (DATE)
- created_at (DATETIME)

## Kitchen_Items
- id (INTEGER PRIMARY KEY)
- name (TEXT)
- category (TEXT: 'utensil', 'consumable')
- sku (TEXT)
- unit (TEXT: 'piece', 'kg', 'liter', 'box')
- current_stock (INTEGER)
- minimum_stock (INTEGER)
- location_id (INTEGER FOREIGN KEY)
- created_at (DATETIME)
- updated_at (DATETIME)

## Kitchen_Purchases
- id (INTEGER PRIMARY KEY)
- item_id (INTEGER FOREIGN KEY)
- supplier_name (TEXT)
- quantity (INTEGER)
- unit_price (DECIMAL)
- total_cost (DECIMAL)
- purchase_date (DATE)
- invoice_number (TEXT)
- location_id (INTEGER FOREIGN KEY)
- created_at (DATETIME)

## Kitchen_Deliveries
- id (INTEGER PRIMARY KEY)
- item_id (INTEGER FOREIGN KEY)
- catering_unit_name (TEXT)
- quantity (INTEGER)
- delivery_date (DATE)
- received_by (TEXT)
- notes (TEXT)
- location_id (INTEGER FOREIGN KEY)
- created_at (DATETIME)

## Uniform_Items
- id (INTEGER PRIMARY KEY)
- name (TEXT)
- category (TEXT: 'shirt', 'pants', 'apron', 'cap', 'shoes')
- size (TEXT)
- gender (TEXT: 'male', 'female', 'unisex')
- current_stock (INTEGER)
- minimum_stock (INTEGER)
- location_id (INTEGER FOREIGN KEY)
- created_at (DATETIME)
- updated_at (DATETIME)

## Uniform_Purchases
- id (INTEGER PRIMARY KEY)
- item_id (INTEGER FOREIGN KEY)
- supplier_name (TEXT)
- quantity (INTEGER)
- unit_price (DECIMAL)
- total_cost (DECIMAL)
- purchase_date (DATE)
- invoice_number (TEXT)
- location_id (INTEGER FOREIGN KEY)
- created_at (DATETIME)

## Uniform_Distributions
- id (INTEGER PRIMARY KEY)
- item_id (INTEGER FOREIGN KEY)
- staff_name (TEXT)
- staff_id (TEXT)
- department (TEXT)
- quantity (INTEGER)
- distribution_date (DATE)
- notes (TEXT)
- location_id (INTEGER FOREIGN KEY)
- created_at (DATETIME)

## Employees
- id (INTEGER PRIMARY KEY)
- employee_id (TEXT UNIQUE)
- first_name (TEXT)
- last_name (TEXT)
- email (TEXT)
- phone (TEXT)
- department (TEXT)
- position (TEXT)
- location_id (INTEGER FOREIGN KEY)
- hire_date (DATE)
- status (TEXT: 'active', 'inactive', 'on_leave')
- address (TEXT)
- emergency_contact (TEXT)
- emergency_phone (TEXT)
- created_at (DATETIME)
- updated_at (DATETIME)
